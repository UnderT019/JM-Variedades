<script>
let metodoSelecionado = '';

function selecionarPagamento(tipo, elemento) {
  metodoSelecionado = tipo;
  document.querySelectorAll('.opcao').forEach(el => el.classList.remove('selecionada'));
  elemento.classList.add('selecionada');

  const form = document.getElementById('form-cartao');
  form.style.display = (tipo === 'cartao') ? 'block' : 'none';
}

function confirmarPagamento() {
  const msg = document.getElementById('msg');
  const selecionado = document.querySelector('input[name="pagamento"]:checked');

  if (!selecionado) {
    msg.innerHTML = '❌ Selecione uma forma de pagamento.';
    msg.className = 'msg erro';
    return;
  }
  msg.className = 'msg'; // limpa erro

  // Textos por método de pagamento
  const textos = {
    cartao: 'Validando dados do cartão de crédito...',
    boleto: 'Gerando boleto bancário...',
    pix: 'Gerando código Pix...'
  };
  document.getElementById('txt-metodo').textContent = textos[metodoSelecionado];

  // Abre o modal na etapa de checagem
  document.getElementById('etapa-checando').style.display = 'block';
  document.getElementById('etapa-sucesso').style.display = 'none';
  document.getElementById('modal').classList.add('ativo');

  // Após 2.5 segundos, mostra o resultado
  setTimeout(() => {
    document.getElementById('etapa-checando').style.display = 'none';
    document.getElementById('etapa-sucesso').style.display = 'block';

    // Status conforme o método (cartão = Pago / boleto e pix = Aguardando)
    const status = document.getElementById('status-final');
    if (metodoSelecionado === 'cartao') {
      status.textContent = 'Pago';
      status.style.color = 'var(--verde)';
    } else {
      status.textContent = 'Aguardando Pagamento';
      status.style.color = 'var(--dourado)';
    }
  }, 2500);
}

function fecharModal() {
  document.getElementById('modal').classList.remove('ativo');
  alert('Pedido concluído com sucesso!');
  window.location.href = 'index.html';
}

// Sincroniza frete e endereço com o módulo logístico
window.addEventListener('DOMContentLoaded', () => {
  document.getElementById('form-cartao').style.display = 'none';

  if (typeof store !== 'undefined') {
    const frete = store.getFreteSelecionado();
    const cotacao = store.getFreteCotacao();
    if (frete) {
      const nomeEl = document.getElementById('txtNomeFreteCheckout');
      const valEl = document.getElementById('txtValorFreteCheckout');
      const totEl = document.getElementById('txtTotalCheckout');
      if (nomeEl) nomeEl.textContent = frete.nome;
      if (valEl) valEl.textContent = frete.valor === 0 ? '🎉 Grátis' : 'R$ ' + frete.valor.toFixed(2).replace('.', ',');
      if (totEl) totEl.textContent = 'R$ ' + (309.70 + frete.valor).toFixed(2).replace('.', ',');
    }
    if (cotacao && cotacao.endereco) {
      const endEl = document.getElementById('enderecoEntregaCheckout');
      if (endEl) {
        endEl.innerHTML = `
          Thiago Silva<br>
          ${cotacao.endereco.logradouro || 'Rua das Palmeiras, 245'}<br>
          ${cotacao.endereco.cidade} - ${cotacao.endereco.uf}<br>
          CEP: ${cotacao.cep}
        `;
      }
    }
  }
});
</script>
