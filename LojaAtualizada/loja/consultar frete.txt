<script>
// ===== SCRIPT DE CÁLCULO DE FRETE LOGÍSTICO =====
let freteSelecionado = 0;
let cotacaoLogistica = null;

function moeda(v){ return 'R$ ' + Number(v || 0).toFixed(2).replace('.',','); }

function mostrarMsg(t, tipo){
  let m = document.getElementById('msg');
  m.innerHTML = t;
  m.className = 'msg ' + tipo;
}

function invalidar(msg){
  freteSelecionado = 0;
  document.getElementById('valorFrete').textContent = '—';
  document.getElementById('opcoes').style.display = 'none';
  document.querySelectorAll('input[name="frete_opcao"]').forEach(r => r.checked = false);
  document.querySelectorAll('.opcao').forEach(o => o.classList.remove('selecionada'));
  atualizarResumo();
  mostrarMsg(msg, 'aviso');
}

async function calcularFrete(){
  let cep = document.getElementById('cep').value.replace(/\D/g,'');
  if(cep.length !== 8){
    document.getElementById('opcoes').style.display = 'none';
    mostrarMsg('❌ CEP inválido. Digite 8 dígitos (ex.: 01001-000).', 'erro');
    return;
  }

  const pesoKg = parseFloat(document.getElementById('pesoSimulacao').value) || 1.3;
  const subtotal = parseFloat(document.getElementById('subtotalSimulacao').value) || 107.80;

  mostrarMsg('⏳ Consultando prazos e tarifas para o CEP <b>' + cep.replace(/(\d{5})(\d{3})/,'$1-$2') + '</b>...', 'aviso');

  const cotacao = await FreteLogistica.calcularCotacao({
    cep: cep,
    pesoKg: pesoKg,
    subtotal: subtotal
  });

  if(!cotacao.sucesso){
    document.getElementById('opcoes').style.display = 'none';
    mostrarMsg(`❌ ${cotacao.msg}`, 'erro');
    return;
  }

  cotacaoLogistica = cotacao;

  const end = cotacao.endereco;
  const localTxt = (end && end.cidade) ? `${end.cidade} - ${end.uf}` : cotacao.regiao;
  const ruaTxt = (end && end.logradouro) ? `${end.logradouro} · ` : '';

  mostrarMsg(`
    ✅ Opções para o CEP <b>${cotacao.cep}</b> (${localTxt}):<br>
    <small style="opacity:0.9;">${ruaTxt}Distância aprox.: ~${cotacao.distanciaKm} km | Peso: ${cotacao.pesoFormatado}</small>
    ${cotacao.elegivelFreteGratis ? `<div style="color:var(--verde); font-weight:800; margin-top:2px;">🎉 Frete Grátis disponível no PAC!</div>` : ''}
  `, 'sucesso');

  const containerOpcoes = document.getElementById('opcoes');
  let optPadrao = cotacao.opcoes.find(o => o.isGratis) || cotacao.opcoes[0];
  freteSelecionado = optPadrao.valor;

  containerOpcoes.innerHTML = FreteLogistica.gerarHtmlOpcoes(cotacao, optPadrao.id, 'selecionarFrete');
  containerOpcoes.style.display = 'block';

  selecionarFrete(optPadrao.id, optPadrao.valor, document.querySelector(`.opcao[data-frete-id="${optPadrao.id}"]`));
}

function selecionarFrete(id, val, el){
  freteSelecionado = Number(val) || 0;
  document.querySelectorAll('.opcao').forEach(o => o.classList.remove('selecionada'));
  if(el){
    el.classList.add('selecionada');
    const radio = el.querySelector('input[type="radio"]');
    if(radio) radio.checked = true;
  }
  document.getElementById('valorFrete').textContent = freteSelecionado === 0 ? '🎉 Grátis' : moeda(freteSelecionado);
  atualizarResumo();
}

function atualizarResumo(){
  const subtotal = parseFloat(document.getElementById('subtotalSimulacao').value) || 0;
  const peso = parseFloat(document.getElementById('pesoSimulacao').value) || 0;
  document.getElementById('subtotal').textContent = moeda(subtotal);
  document.getElementById('peso').textContent = peso.toFixed(1).replace('.',',') + ' kg';
  document.getElementById('total').textContent = moeda(subtotal + freteSelecionado);
}

function finalizar(){
  if(freteSelecionado === 0 && !cotacaoLogistica){
    mostrarMsg('❌ Calcule e selecione uma opção de frete para continuar.', 'erro');
    return;
  }
  alert('Opção de frete selecionada com sucesso! Redirecionando para o checkout.');
  window.location.href = 'index.html';
}

// Máscara automática do CEP
document.getElementById('cep').addEventListener('input', e => {
  let v = e.target.value.replace(/\D/g,'').slice(0,8);
  e.target.value = v.replace(/(\d{5})(\d)/,'$1-$2');
});

document.getElementById('pesoSimulacao').addEventListener('change', () => {
  if(cotacaoLogistica) calcularFrete();
  else atualizarResumo();
});
document.getElementById('subtotalSimulacao').addEventListener('change', () => {
  if(cotacaoLogistica) calcularFrete();
  else atualizarResumo();
});

atualizarResumo();
</script>
